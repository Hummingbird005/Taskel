' Taskel for Outlook: the window that asks who gets the email.
'
' This is the code of a UserForm named TaskelSendForm. In the VBA editor:
' Insert > UserForm, set its (Name) to TaskelSendForm in the Properties
' window, open its code (F7), delete what is there and paste this file.
' The controls are created by the code, so the form stays empty in the
' designer. README.md next to this file has the full steps.
'
' The macro (TaskelOutlook) calls Prepare, shows the form, then reads Sent,
' RecipientsJson and Instructions. Without this form, every email becomes a
' task of your own, as before.
'
' The look follows Taskel: Segoe UI, white surfaces, thin grey borders and
' one blue accent. VBA forms have no rounded corners or shadows; flat
' controls with single-pixel borders come closest.
'
' Plain ASCII on purpose: Romanian letters are written as \uXXXX and turned
' into characters by U() when the form opens.

Option Explicit

Private Const MAX_RECIPIENTS As Long = 50
Private Const MAX_INSTRUCTIONS As Long = 2000

' Layout, in points.
Private Const FORM_WIDTH As Single = 400
Private Const PAD As Single = 16
Private Const HEADER_HEIGHT As Single = 58
Private Const FOOTER_TOP As Single = 410
Private Const FOOTER_HEIGHT As Single = 50

' MSForms values, written out so the code does not depend on their names.
Private Const FLAT As Long = 0              ' fmSpecialEffectFlat
Private Const SINGLE_BORDER As Long = 1     ' fmBorderStyleSingle
Private Const NO_BORDER As Long = 0         ' fmBorderStyleNone
Private Const OPAQUE As Long = 1            ' fmBackStyleOpaque
Private Const TRANSPARENT As Long = 0       ' fmBackStyleTransparent
Private Const MULTI_SELECT As Long = 1      ' fmMultiSelectMulti
Private Const CHECK_LIST As Long = 1        ' fmListStyleOption
Private Const VERTICAL_BAR As Long = 2      ' fmScrollBarsVertical

Private WithEvents txtSearch As MSForms.TextBox
Private WithEvents lblPlaceholder As MSForms.Label
Private WithEvents lstPeople As MSForms.ListBox
Private WithEvents btnSend As MSForms.CommandButton
Private WithEvents btnCancel As MSForms.CommandButton
Private WithEvents lblFooter As MSForms.Label
Private txtNotes As MSForms.TextBox
Private lblSubtitle As MSForms.Label
Private lblChosen As MSForms.Label

' Everyone who can get the email, as Taskel listed them, and who is ticked.
Private entryValues() As String
Private entryNames() As String
Private entryDetails() As String
Private entryChosen() As Boolean
Private entryCount As Long
' Which entry each row of the list shows, as the search narrows it.
Private rowEntry() As Long
Private filling As Boolean
Private wasSent As Boolean

' ---------------------------------------------------------------------------
' What the macro uses
' ---------------------------------------------------------------------------

' recipients: Taskel's answer to GET /api/outlook/recipients, one line per
' person or group: kind TAB value TAB name TAB detail.
Public Sub Prepare(ByVal recipients As String, ByVal mailCount As Long)
    Dim lines() As String
    Dim fields() As String
    Dim i As Long

    lines = Split(Replace(recipients, vbCr, ""), vbLf)
    entryCount = 0
    ReDim entryValues(0 To UBound(lines) + 1)
    ReDim entryNames(0 To UBound(lines) + 1)
    ReDim entryDetails(0 To UBound(lines) + 1)
    ReDim entryChosen(0 To UBound(lines) + 1)
    ReDim rowEntry(0 To UBound(lines) + 1)

    For i = 0 To UBound(lines)
        fields = Split(lines(i), vbTab)
        If UBound(fields) >= 3 Then
            entryValues(entryCount) = fields(1)
            entryNames(entryCount) = fields(2)
            entryDetails(entryCount) = fields(3)
            entryCount = entryCount + 1
        End If
    Next i

    If mailCount = 1 Then
        lblSubtitle.Caption = U("1 email selectat \u00b7 fiecare destinatar prime\u0219te task-ul lui")
    Else
        lblSubtitle.Caption = mailCount & U(" emailuri selectate \u00b7 fiecare destinatar prime\u0219te task-urile lui")
    End If
    ShowMatches
    UpdateChosen
End Sub

' True when the user clicked Trimite, False when they gave up.
Public Property Get Sent() As Boolean
    Sent = wasSent
End Property

' The ticked people and groups, as the JSON array Taskel takes: ["id","group:id"].
' Empty ([]) when nobody was ticked: the task is then the user's own.
Public Property Get RecipientsJson() As String
    Dim result As String
    Dim i As Long

    For i = 0 To entryCount - 1
        If entryChosen(i) Then
            If Len(result) > 0 Then result = result & ","
            ' Ids and "group:" ids only: nothing in them needs escaping.
            result = result & Chr$(34) & entryValues(i) & Chr$(34)
        End If
    Next i
    RecipientsJson = "[" & result & "]"
End Property

Public Property Get Instructions() As String
    Instructions = Trim$(txtNotes.Text)
End Property

' ---------------------------------------------------------------------------
' The window
' ---------------------------------------------------------------------------

Private Sub UserForm_Initialize()
    Dim lbl As MSForms.Label
    Dim inner As Single

    inner = FORM_WIDTH - 2 * PAD
    Me.Caption = "Taskel"
    Me.BackColor = ColorWhite()
    Me.Font.Name = "Segoe UI"
    Me.Font.Size = 9
    Me.Width = FORM_WIDTH + 6
    Me.Height = FOOTER_TOP + FOOTER_HEIGHT + 26

    ' Header: title and what is being sent.
    Set lbl = AddLabel("lblTitle", U("Trimite \u00een Taskel"), PAD, 12, inner, 22)
    lbl.Font.Size = 13
    lbl.Font.Bold = True
    lbl.ForeColor = ColorText()
    Set lblSubtitle = AddLabel("lblSubtitle", "", PAD, 34, inner, 14)
    lblSubtitle.ForeColor = ColorMuted()
    AddLine "lineHeader", HEADER_HEIGHT

    ' Who gets it.
    AddSectionTitle "lblTo", U("CUI TRIMI\u021aI"), 72
    Set txtSearch = Me.Controls.Add("Forms.TextBox.1", "txtSearch")
    Place txtSearch, PAD, 88, inner, 24
    StyleField txtSearch
    Set lblPlaceholder = AddLabel("lblPlaceholder", U("Caut\u0103 o persoan\u0103 sau un grup"), PAD + 7, 93, inner - 14, 14)
    lblPlaceholder.ForeColor = ColorMuted()
    lblPlaceholder.BackStyle = TRANSPARENT

    Set lstPeople = Me.Controls.Add("Forms.ListBox.1", "lstPeople")
    Place lstPeople, PAD, 118, inner, 150
    With lstPeople
        .Font.Name = "Segoe UI"
        .Font.Size = 10
        .ForeColor = ColorText()
        .BackColor = ColorWhite()
        .SpecialEffect = FLAT
        .BorderStyle = SINGLE_BORDER
        .BorderColor = ColorBorder()
        .ColumnCount = 2
        .ColumnWidths = "190 pt;160 pt"
        .MultiSelect = MULTI_SELECT
        .ListStyle = CHECK_LIST
        .IntegralHeight = False
    End With

    Set lblChosen = AddLabel("lblChosen", "", PAD, 274, inner, 28)
    lblChosen.WordWrap = True

    ' What they should do.
    Set lbl = AddSectionTitle("lblNotes", U("INDICA\u021aII"), 310)
    Set lbl = AddLabel("lblOptional", U("op\u021bional, le vede fiecare destinatar"), PAD + 54, 310, inner - 54, 14)
    lbl.ForeColor = ColorMuted()
    Set txtNotes = Me.Controls.Add("Forms.TextBox.1", "txtNotes")
    Place txtNotes, PAD, 326, inner, 70
    StyleField txtNotes
    With txtNotes
        .MultiLine = True
        .WordWrap = True
        .EnterKeyBehavior = True
        .ScrollBars = VERTICAL_BAR
        .MaxLength = MAX_INSTRUCTIONS
    End With

    ' Footer: a grey band with the two buttons on the right.
    Set lblFooter = AddLabel("lblFooter", "", 0, FOOTER_TOP, FORM_WIDTH + 6, FOOTER_HEIGHT)
    lblFooter.BackStyle = OPAQUE
    lblFooter.BackColor = ColorSurface()
    AddLine "lineFooter", FOOTER_TOP

    Set btnCancel = Me.Controls.Add("Forms.CommandButton.1", "btnCancel")
    Place btnCancel, FORM_WIDTH - PAD - 96 - 8 - 80, FOOTER_TOP + 11, 80, 28
    StyleButton btnCancel, False
    btnCancel.Caption = U("Renun\u021b\u0103")
    btnCancel.Cancel = True

    Set btnSend = Me.Controls.Add("Forms.CommandButton.1", "btnSend")
    Place btnSend, FORM_WIDTH - PAD - 96, FOOTER_TOP + 11, 96, 28
    StyleButton btnSend, True
    btnSend.Caption = "Trimite"
    ' Not the default button: Enter in the search box ticks a person instead of sending.

    txtSearch.TabIndex = 0
    lstPeople.TabIndex = 1
    txtNotes.TabIndex = 2
    btnSend.TabIndex = 3
    btnCancel.TabIndex = 4
End Sub

' Closing the window with X counts as Renunta. The form is hidden, not
' unloaded, so the macro can still read that nothing was sent.
Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    If CloseMode = vbFormControlMenu Then
        Cancel = True
        wasSent = False
        Me.Hide
    End If
End Sub

Private Sub txtSearch_Change()
    lblPlaceholder.Visible = (Len(txtSearch.Text) = 0)
    ShowMatches
End Sub

' Enter ticks the first person found; the down arrow moves into the list.
Private Sub txtSearch_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    If KeyCode = vbKeyReturn And lstPeople.ListCount > 0 And Len(txtSearch.Text) > 0 Then
        lstPeople.Selected(0) = Not lstPeople.Selected(0)
        txtSearch.Text = ""
        KeyCode = 0
    ElseIf KeyCode = vbKeyDown And lstPeople.ListCount > 0 Then
        lstPeople.SetFocus
        lstPeople.ListIndex = 0
        KeyCode = 0
    End If
End Sub

Private Sub lblPlaceholder_Click()
    txtSearch.SetFocus
End Sub

Private Sub lstPeople_Change()
    Dim row As Long

    If filling Then Exit Sub
    For row = 0 To lstPeople.ListCount - 1
        entryChosen(rowEntry(row)) = lstPeople.Selected(row)
    Next row
    UpdateChosen
End Sub

Private Sub btnSend_Click()
    If ChosenCount() > MAX_RECIPIENTS Then
        MsgBox U("Po\u021bi trimite la cel mult ") & MAX_RECIPIENTS & U(" destinatari odat\u0103."), _
            vbExclamation, "Taskel"
        Exit Sub
    End If
    wasSent = True
    Me.Hide
End Sub

Private Sub btnCancel_Click()
    wasSent = False
    Me.Hide
End Sub

' A darker blue while the mouse is over Trimite, as on a web page.
Private Sub btnSend_MouseMove(ByVal Button As Integer, ByVal Shift As Integer, ByVal X As Single, ByVal Y As Single)
    If btnSend.BackColor <> ColorAccentHover() Then btnSend.BackColor = ColorAccentHover()
End Sub

Private Sub btnCancel_MouseMove(ByVal Button As Integer, ByVal Shift As Integer, ByVal X As Single, ByVal Y As Single)
    If btnCancel.BackColor <> ColorSurface() Then btnCancel.BackColor = ColorSurface()
    ResetHover btnSend, ColorAccent()
End Sub

Private Sub lblFooter_MouseMove(ByVal Button As Integer, ByVal Shift As Integer, ByVal X As Single, ByVal Y As Single)
    ResetHover btnSend, ColorAccent()
    ResetHover btnCancel, ColorWhite()
End Sub

Private Sub UserForm_MouseMove(ByVal Button As Integer, ByVal Shift As Integer, ByVal X As Single, ByVal Y As Single)
    ResetHover btnSend, ColorAccent()
    ResetHover btnCancel, ColorWhite()
End Sub

' ---------------------------------------------------------------------------
' Helpers
' ---------------------------------------------------------------------------

' Fills the list with the entries whose name or detail contains the search
' text, ignoring case and Romanian diacritics, and keeps what was ticked.
Private Sub ShowMatches()
    Dim search As String
    Dim i As Long
    Dim row As Long

    search = Plain(Trim$(txtSearch.Text))
    filling = True
    lstPeople.Clear
    row = 0
    For i = 0 To entryCount - 1
        If Len(search) = 0 Or InStr(1, Plain(entryNames(i) & " " & entryDetails(i)), search, vbBinaryCompare) > 0 Then
            lstPeople.AddItem entryNames(i)
            lstPeople.List(row, 1) = entryDetails(i)
            lstPeople.Selected(row) = entryChosen(i)
            rowEntry(row) = i
            row = row + 1
        End If
    Next i
    filling = False
End Sub

' Says who is ticked, also those the search hides, and names the button after it.
Private Sub UpdateChosen()
    Dim chosen As Long
    Dim names As String
    Dim i As Long

    For i = 0 To entryCount - 1
        If entryChosen(i) Then
            chosen = chosen + 1
            If Len(names) < 160 Then
                If Len(names) > 0 Then names = names & ", "
                names = names & entryNames(i)
            End If
        End If
    Next i

    If chosen = 0 Then
        lblChosen.ForeColor = ColorMuted()
        lblChosen.Caption = U("Niciun destinatar ales: task-ul r\u0103m\u00e2ne la tine.")
        btnSend.Caption = "Trimite"
    Else
        If Len(names) >= 160 Then names = names & ", ..."
        lblChosen.ForeColor = ColorAccent()
        lblChosen.Caption = U("Ale\u0219i (") & chosen & "): " & names & U(". R\u0103spunsurile vin la tine.")
        If chosen = 1 Then
            btnSend.Caption = "Trimite"
        Else
            btnSend.Caption = "Trimite (" & chosen & ")"
        End If
    End If
End Sub

Private Function ChosenCount() As Long
    Dim i As Long

    For i = 0 To entryCount - 1
        If entryChosen(i) Then ChosenCount = ChosenCount + 1
    Next i
End Function

    lbl.BackStyle = TRANSPARENT
    lbl.ForeColor = ColorText()

Private Sub StyleField(ByVal box As MSForms.TextBox)
    box.Font.Name = "Segoe UI"
    box.Font.Size = 10
    box.ForeColor = ColorText()
    box.BackColor = ColorWhite()
    box.SpecialEffect = FLAT
    box.BorderStyle = SINGLE_BORDER
    box.BorderColor = ColorField()

Private Sub StyleButton(ByVal button As MSForms.CommandButton, ByVal primary As Boolean)
    button.Font.Name = "Segoe UI"
    button.Font.Size = 9
    button.Font.Bold = primary
    button.TakeFocusOnClick = False
    If primary Then
        button.BackColor = ColorAccent()
        button.ForeColor = ColorWhite()
    Else
        button.BackColor = ColorWhite()
        button.ForeColor = ColorText()
    End If
End Sub

Private Sub ResetHover(ByVal button As MSForms.CommandButton, ByVal color As Long)
    If button.BackColor <> color Then button.BackColor = color
End Sub

                  ByVal widthPts As Single, ByVal heightPts As Single)
    ctl.Left = leftPos
    ctl.Top = topPos
    ctl.Width = widthPts

' Lower case, without Romanian diacritics, so "stefan" finds a name written with the letter s-comma.
Private Function Plain(ByVal value As String) As String
    Dim result As String

    result = value
    result = Replace(result, ChrW(&H102), "a")
    result = Replace(result, ChrW(&HC2), "a")
    result = Replace(result, ChrW(&HCE), "i")
    result = Replace(result, ChrW(&H218), "s")
    result = Replace(result, ChrW(&H15E), "s")
    result = Replace(result, ChrW(&H21A), "t")
    result = Replace(result, ChrW(&H162), "t")
    result = LCase$(result)
    result = Replace(result, ChrW(&H103), "a")
    result = Replace(result, ChrW(&HE2), "a")
    result = Replace(result, ChrW(&HEE), "i")
    result = Replace(result, ChrW(&H219), "s")
    result = Replace(result, ChrW(&H15F), "s")
    result = Replace(result, ChrW(&H21B), "t")
    result = Replace(result, ChrW(&H163), "t")
    Plain = result
End Function

' Turns each \uXXXX in text into the character it names.
Private Function U(ByVal text As String) As String
    Dim result As String
    Dim i As Long

    i = 1
    Do While i <= Len(text)
        If Mid$(text, i, 2) = "\u" And i + 5 <= Len(text) Then
            result = result & ChrW(CLng("&H" & Mid$(text, i + 2, 4)))
            i = i + 6
        Else
            result = result & Mid$(text, i, 1)
            i = i + 1
        End If
    Loop
    U = result
End Function
