' Taskel for Outlook: the window that asks who gets the email, and by when.
'
' This is the code of a UserForm named TaskelSendForm. In the VBA editor:
' Insert > UserForm, set its (Name) to TaskelSendForm in the Properties
' window, open its code (F7), delete what is there and paste this file.
' The controls are created by the code, so the form stays empty in the
' designer. README.md next to this file has the full steps.
'
' The macro (TaskelOutlook) calls Prepare, shows the form, then reads Sent,
' RecipientsJson, Instructions and DueDate. Without this form, every email
' becomes a task of your own, as before.
'
' The "Catre" field works like the To field of an email: type part of a name,
' pick a suggestion with the mouse or the arrow keys and Enter, and names are
' kept as "Andrei Stan; Ofertare; ". Backspace at the end removes the last one.
'
' The look follows Taskel: Segoe UI, white surfaces, thin grey borders and
' one blue accent. VBA forms have no rounded corners or shadows; flat
' controls with single-pixel borders come closest.
'
' Plain ASCII on purpose: Romanian letters are written as \uXXXX and turned
' into characters by U() when the form opens.

' recipients: Taskel's answer to GET /api/outlook/recipients, one line per
' person or group: kind TAB value TAB name TAB detail.
Public Sub Prepare(ByVal recipients As String, ByVal mailCount As Long)
    Dim lines() As String
    Dim fields() As String
    Dim size As Long
    Dim i As Long
    Dim j As Long

    lines = Split(Replace(recipients, vbCr, ""), vbLf)
    size = UBound(lines) + 1
    ReDim entryValues(0 To size)
    ReDim entryNames(0 To size)
    ReDim entryDetails(0 To size)
    ReDim entryEmails(0 To size)
    ReDim entryTokens(0 To size)
    ReDim chosen(0 To size)
    ReDim suggested(0 To MAX_SUGGESTIONS)
    entryCount = 0

    For i = 0 To UBound(lines)
        fields = Split(lines(i), vbTab)
        If UBound(fields) >= 3 Then
            entryValues(entryCount) = fields(1)
            entryNames(entryCount) = Trim$(fields(2))
            entryDetails(entryCount) = fields(3)
            If fields(0) = "person" Then entryEmails(entryCount) = Trim$(Replace(fields(3), "(eu)", ""))
            entryCount = entryCount + 1
        End If
    Next i

    ' Two entries with the same name are told apart by the address, or as the group.
    For i = 0 To entryCount - 1
        entryTokens(i) = entryNames(i)
        For j = 0 To entryCount - 1
            If j <> i And Plain(entryNames(j)) = Plain(entryNames(i)) Then
                If Len(entryEmails(i)) > 0 Then
                    entryTokens(i) = entryNames(i) & " <" & entryEmails(i) & ">"
                Else
                    entryTokens(i) = entryNames(i) & " (grup)"
                End If
                Exit For
            End If
        Next j
    Next i

    If mailCount = 1 Then
        lblSubtitle.Caption = U("1 email selectat \u00b7 fiecare destinatar prime\u0219te task-ul lui")
    Else
        lblSubtitle.Caption = mailCount & U(" emailuri selectate \u00b7 fiecare destinatar prime\u0219te task-urile lui")
    End If
    UpdateToHint
    UpdateDueHint
End Sub

' True when the user clicked Trimite, False when they gave up.
Public Property Get Sent() As Boolean
    Sent = wasSent
End Property

' The people and groups in the field, as the JSON array Taskel takes: ["id","group:id"].
' Empty ([]) when the field was left empty: the task is then the user's own.
Public Property Get RecipientsJson() As String
    Dim result As String
    Dim i As Long

    For i = 0 To chosenCount - 1
        If Len(result) > 0 Then result = result & ","
        ' Ids and "group:" ids only: nothing in them needs escaping.
        result = result & Chr$(34) & entryValues(chosen(i)) & Chr$(34)
    Next i
    RecipientsJson = "[" & result & "]"
End Property

Public Property Get Instructions() As String
    Instructions = Trim$(txtNotes.Text)
End Property

' The due date as yyyy-mm-dd, or "" for none.
Public Property Get DueDate() As String
    DueDate = dueValue
End Property

' --- The To field --------------------------------------------------------

Private Sub txtTo_Change()
    If Not ready Or updating Then Exit Sub
    RefreshSuggestions
    UpdateToHint
End Sub

Private Sub txtTo_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    If Not ready Then Exit Sub
    Select Case KeyCode
        Case vbKeyDown
            If lstSuggest.Visible Then
                If lstSuggest.ListIndex < suggestedCount - 1 Then lstSuggest.ListIndex = lstSuggest.ListIndex + 1
                KeyCode = 0
            End If
        Case vbKeyUp
            If lstSuggest.Visible Then
                If lstSuggest.ListIndex > 0 Then lstSuggest.ListIndex = lstSuggest.ListIndex - 1
                KeyCode = 0
            End If
        Case vbKeyReturn
            ' Enter picks the highlighted suggestion; the field never takes a line break.
            If lstSuggest.Visible Then AcceptSuggestion
            KeyCode = 0
        Case vbKeyTab
            HideSuggestions
        Case vbKeyEscape
            If lstSuggest.Visible Then
                HideSuggestions
                KeyCode = 0
            End If
        Case vbKeyBack
            If txtTo.SelLength = 0 And txtTo.SelStart >= Len(RTrim$(txtTo.Text)) Then
                If RemoveLastName() Then KeyCode = 0
            End If
    End Select
End Sub

Private Sub lstSuggest_MouseUp(ByVal Button As Integer, ByVal Shift As Integer, ByVal X As Single, ByVal Y As Single)
    If ready Then AcceptSuggestion
End Sub

Private Sub lstSuggest_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    If Not ready Then Exit Sub
    If KeyCode = vbKeyReturn Then
        AcceptSuggestion
        KeyCode = 0
    ElseIf KeyCode = vbKeyEscape Then
        HideSuggestions
        txtTo.SetFocus
        KeyCode = 0
    End If
End Sub

    HideSuggestions
    If Not ResolveRecipients() Then

        UpdateDueHint

        Exit Sub
    End If
    wasSent = True
    Me.Hide
End Sub

' The name being typed: whatever follows the last semicolon.
Private Function Fragment() As String
    Dim t As String

    t = txtTo.Text
    Fragment = Trim$(Mid$(t, InStrRev(t, ";") + 1))
End Function

' The entry a name in the field stands for; -1 when none, -2 when the name is
' shared by several and was written without the address.
Private Function FindEntry(ByVal token As String) As Long
    Dim key As String
    Dim hits As Long
    Dim i As Long

    FindEntry = -1
    key = Plain(Trim$(token))
    If Len(key) = 0 Then Exit Function
    For i = 0 To entryCount - 1
        If Plain(entryTokens(i)) = key Then
            FindEntry = i
            Exit Function
        End If
    Next i
    For i = 0 To entryCount - 1
        If Len(entryEmails(i)) > 0 Then
            If Plain(entryEmails(i)) = key Then
                FindEntry = i
                Exit Function
            End If
        End If
    Next i
    For i = 0 To entryCount - 1
        If Plain(entryNames(i)) = key Then hits = hits + 1
    Next i
    If hits > 1 Then FindEntry = -2
End Function

' The entries already written in full in the field, as ";3;7;", so they are not suggested again.
Private Function TakenKeys() As String
    Dim parts As Variant
    Dim found As Long
    Dim i As Long

    TakenKeys = ";"
    parts = Split(txtTo.Text, ";")
    For i = 0 To UBound(parts) - 1
        found = FindEntry(CStr(parts(i)))
        If found >= 0 Then TakenKeys = TakenKeys & found & ";"
    Next i
End Function

' Lists the people and groups whose name or address contains what is being
' typed, ignoring case and Romanian diacritics.
Private Sub RefreshSuggestions()
    Dim search As String
    Dim taken As String
    Dim i As Long

    search = Plain(Fragment())
    suggestedCount = 0
    lstSuggest.Clear
    If Len(search) > 0 Then
        taken = TakenKeys()
        For i = 0 To entryCount - 1
            If InStr(1, taken, ";" & i & ";", vbBinaryCompare) = 0 Then
                If InStr(1, Plain(entryNames(i) & " " & entryDetails(i)), search, vbBinaryCompare) > 0 Then
                    lstSuggest.AddItem entryNames(i)
                    lstSuggest.List(suggestedCount, 1) = entryDetails(i)
                    suggested(suggestedCount) = i
                    suggestedCount = suggestedCount + 1
                    If suggestedCount = MAX_SUGGESTIONS Then Exit For
                End If
            End If
        Next i
    End If

    If suggestedCount > 0 Then
        lstSuggest.Height = suggestedCount * ROW_HEIGHT + 4
        lstSuggest.ListIndex = 0
        lstSuggest.Visible = True
    Else
        lstSuggest.Visible = False
    End If
End Sub

Private Sub HideSuggestions()
    If lstSuggest.Visible Then lstSuggest.Visible = False
End Sub

' Replaces the name being typed with the highlighted suggestion, followed by "; ".
Private Sub AcceptSuggestion()
    Dim t As String
    Dim base As String

    If lstSuggest.ListIndex < 0 Or lstSuggest.ListIndex >= suggestedCount Then Exit Sub
    t = txtTo.Text
    base = RTrim$(Left$(t, InStrRev(t, ";")))
    If Len(base) > 0 Then base = base & " "
    updating = True
    txtTo.Text = base & entryTokens(suggested(lstSuggest.ListIndex)) & "; "
    updating = False
    HideSuggestions
    txtTo.SetFocus
    txtTo.SelStart = Len(txtTo.Text)
    UpdateToHint
End Sub

' Backspace right after "; " removes the whole last name, as in Outlook.
Private Function RemoveLastName() As Boolean
    Dim t As String

    t = RTrim$(txtTo.Text)
    If Len(t) = 0 Or Len(Fragment()) > 0 Then Exit Function
    If Right$(t, 1) = ";" Then t = Left$(t, Len(t) - 1)
    t = RTrim$(Left$(t, InStrRev(t, ";")))
    If Len(t) > 0 Then t = t & " "
    updating = True
    txtTo.Text = t
    updating = False
    txtTo.SelStart = Len(t)
    HideSuggestions
    UpdateToHint
    RemoveLastName = True
End Function

' Says who the email goes to, or which names are not recognised. The name
' still being typed (after the last semicolon) is not judged yet.
Private Sub UpdateToHint()
    Dim parts As Variant
    Dim token As String
    Dim names As Long
    Dim bad As Long
    Dim i As Long

    If Not ready Then Exit Sub
    parts = Split(txtTo.Text, ";")
    For i = 0 To UBound(parts) - 1
        token = Trim$(parts(i))
        If Len(token) > 0 Then
            If FindEntry(token) >= 0 Then names = names + 1 Else bad = bad + 1
        End If
    Next i

    If bad > 0 Then
        lblToHint.ForeColor = ColorDanger()
        If bad = 1 Then
            lblToHint.Caption = U("Un nume nu este recunoscut: alege-l din sugestii sau corecteaz\u0103-l.")
        Else
            lblToHint.Caption = bad & U(" nume nu sunt recunoscute: alege-le din sugestii sau corecteaz\u0103-le.")
        End If
    ElseIf names = 0 Then
        lblToHint.ForeColor = ColorMuted()
        lblToHint.Caption = U("Scrie un nume \u0219i alege din sugestii. F\u0103r\u0103 destinatari, task-ul r\u0103m\u00e2ne la tine.")
    ElseIf names = 1 Then
        lblToHint.ForeColor = ColorAccent()
        lblToHint.Caption = U("1 destinatar: prime\u0219te task-ul, iar r\u0103spunsul vine la tine.")
    Else
        lblToHint.ForeColor = ColorAccent()
        lblToHint.Caption = names & U(" destinatari: fiecare prime\u0219te task-ul lui, iar r\u0103spunsurile vin la tine.")
    End If
    If names > 1 Then btnSend.Caption = "Trimite (" & names & ")" Else btnSend.Caption = "Trimite"
End Sub

' Works out who the email goes to from everything in the field, the name being
' typed included. False, with the reason shown, when a name is not recognised.
Private Function ResolveRecipients() As Boolean
    Dim parts As Variant
    Dim token As String
    Dim unknown As String
    Dim found As Long
    Dim twice As Boolean
    Dim i As Long
    Dim j As Long

    chosenCount = 0
    parts = Split(txtTo.Text, ";")
    For i = 0 To UBound(parts)
        token = Trim$(parts(i))
        If Len(token) > 0 Then
            found = FindEntry(token)
            If found < 0 Then
                If Len(unknown) > 0 Then unknown = unknown & ", "
                unknown = unknown & U("\u201e") & token & U("\u201d")
                If found = -2 Then unknown = unknown & U(" (sunt mai mul\u021bi cu acest nume)")
            Else
                twice = False
                For j = 0 To chosenCount - 1
                    If chosen(j) = found Then twice = True
                Next j
                If Not twice Then
                    chosen(chosenCount) = found
                    chosenCount = chosenCount + 1
                End If
            End If
        End If
    Next i

    If Len(unknown) > 0 Then
        lblToHint.ForeColor = ColorDanger()
        lblToHint.Caption = U("Nu g\u0103sesc: ") & unknown & U(". Alege numele din sugestii.")
        Exit Function
    End If
    If chosenCount > MAX_RECIPIENTS Then
        lblToHint.ForeColor = ColorDanger()
        lblToHint.Caption = U("Po\u021bi trimite la cel mult ") & MAX_RECIPIENTS & U(" destinatari odat\u0103.")
        Exit Function
    End If
    ResolveRecipients = True
End Function

' ---------------------------------------------------------------------------
' Due date
' ---------------------------------------------------------------------------

Private Sub SetDue(ByVal d As Date)
    If Not ready Then Exit Sub
    HideSuggestions
    txtDue.Text = Format$(d, "dd\.mm\.yyyy")
End Sub

' The next Friday; a week from today when today is Friday.
Private Function NextFriday() As Date
    Dim days As Long

    days = (vbFriday - Weekday(Date, vbSunday) + 7) Mod 7
    If days = 0 Then days = 7
    NextFriday = Date + days
End Function

' Reads dd.mm.yyyy (also with / or -, and a two-digit year). False for anything else.
Private Function ParseDue(ByVal text As String, ByRef result As Date) As Boolean
    Dim parts() As String
    Dim dd As Long
    Dim mm As Long
    Dim yy As Long

    text = Replace(Replace(Trim$(text), "/", "."), "-", ".")
    parts = Split(text, ".")
    If UBound(parts) <> 2 Then Exit Function
    If Not (IsDigits(parts(0)) And IsDigits(parts(1)) And IsDigits(parts(2))) Then Exit Function
    dd = CLng(parts(0))
    mm = CLng(parts(1))
    yy = CLng(parts(2))
    If yy < 100 Then yy = yy + 2000
    If dd < 1 Or dd > 31 Or mm < 1 Or mm > 12 Or yy < 2000 Or yy > 2100 Then Exit Function
    result = DateSerial(yy, mm, dd)
    ' DateSerial turns 31.02 into a day in March; such a date was not a real one.
    ParseDue = (Day(result) = dd And Month(result) = mm)
End Function

Private Function IsDigits(ByVal value As String) As Boolean
    Dim i As Long

    value = Trim$(value)
    If Len(value) = 0 Or Len(value) > 4 Then Exit Function
    For i = 1 To Len(value)
        If InStr("0123456789", Mid$(value, i, 1)) = 0 Then Exit Function
    Next i
    IsDigits = True
End Function

' Repeats the chosen date in words, so a typo shows at once.
Private Sub UpdateDueHint()
    Dim d As Date
    Dim days As Long

    If Not ready Then Exit Sub
    If Len(Trim$(txtDue.Text)) = 0 Then
        lblDueHint.ForeColor = ColorMuted()
        lblDueHint.Caption = U("F\u0103r\u0103 termen.")
    ElseIf ParseDue(txtDue.Text, d) Then
        days = DateDiff("d", Date, d)
        lblDueHint.Caption = RoWeekday(d) & ", " & Day(d) & " " & RoMonth(d) & " " & Year(d) & RelativeText(days)
        If days < 0 Then lblDueHint.ForeColor = ColorDanger() Else lblDueHint.ForeColor = ColorMuted()
    Else
        lblDueHint.ForeColor = ColorDanger()
        lblDueHint.Caption = U("Scrie data ca zz.ll.aaaa, de exemplu ") & Format$(Date + 1, "dd\.mm\.yyyy") & "."
    End If
End Sub

Private Function RelativeText(ByVal days As Long) As String
    Select Case days
        Case 0: RelativeText = U(" \u00b7 azi")
        Case 1: RelativeText = U(" \u00b7 m\u00e2ine")
        Case -1: RelativeText = U(" \u00b7 ieri, \u00een trecut")
        Case Is > 1: RelativeText = U(" \u00b7 peste ") & days & " zile"
        Case Else: RelativeText = U(" \u00b7 acum ") & -days & U(" zile, \u00een trecut")
    End Select
End Function

' Lower case, without Romanian diacritics, so "stefan" finds a name written with the letter s-comma.
Private Function Plain(ByVal value As String) As String
    Dim result As String

    result = value
    result = Replace(result, ChrW(&H102), "a")
    result = Replace(result, ChrW(&HC2), "a")
    result = Replace(result, ChrW(&HCE), "i")
    result = Replace(result, ChrW(&H218), "s")
    result = Replace(result, ChrW(&H15E), "s")
    result = Replace(result, ChrW(&H21A), "t")
    result = Replace(result, ChrW(&H162), "t")
    result = LCase$(result)
    result = Replace(result, ChrW(&H103), "a")
    result = Replace(result, ChrW(&HE2), "a")
    result = Replace(result, ChrW(&HEE), "i")
    result = Replace(result, ChrW(&H219), "s")
    result = Replace(result, ChrW(&H15F), "s")
    result = Replace(result, ChrW(&H21B), "t")
    result = Replace(result, ChrW(&H163), "t")
    Plain = result
End Function

' Turns each \uXXXX in text into the character it names.
Private Function U(ByVal text As String) As String
    Dim result As String
    Dim i As Long

    i = 1
    Do While i <= Len(text)
        If Mid$(text, i, 2) = "\u" And i + 5 <= Len(text) Then
            result = result & ChrW(CLng("&H" & Mid$(text, i + 2, 4)))
            i = i + 6
        Else
            result = result & Mid$(text, i, 1)
            i = i + 1
        End If
    Loop
    U = result
End Function
